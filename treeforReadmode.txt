function Get-FtpTree {
     param (
         $url,
         $user,
         $pass,
         $path = "",
         $visited = @{}
     )

     $fullPath = "$url$path"

     if ($visited.ContainsKey($fullPath)) {
         return
     }
     $visited[$fullPath] = $true

     try {
         $request = [System.Net.FtpWebRequest]::Create($fullPath)
         $request.Credentials = New-Object System.Net.NetworkCredential($user,$pass)
        $request.Method = [System.Net.WebRequestMethods+Ftp]::ListDirectory
         $request.KeepAlive = $false

         $response = $request.GetResponse()
         $reader = New-Object System.IO.StreamReader($response.GetResponseStream())
         $lines = $reader.ReadToEnd().Split("`n")

         foreach ($name in $lines) {
             $name = $name.Trim()
            if ($name -eq "") { continue }

             $currentPath = "$path/$name"
             Write-Host $currentPath

             # Try to go inside (if it's a directory)
             try {
                 
                 Get-FtpTree $url $user $pass $currentPath $visited
             }
            catch {}
        }

         $reader.Close()
         $response.Close()
     }
     catch {
         # silently skip files (not directories)
    }
 }
