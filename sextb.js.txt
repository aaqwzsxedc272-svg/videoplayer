var lastScrollTop = 0;
var tickingScroll = false;
window.addEventListener('scroll', function () {
    if (tickingScroll) return;
    tickingScroll = true;

    window.requestAnimationFrame(function () {
        var st = window.pageYOffset || document.documentElement.scrollTop || 0;
        var netword = $(".network-bar");
        var navbar = $(".navbar");

        if (st > lastScrollTop) {
            netword.addClass("network-top-hide");
            navbar.addClass("navbar-hide");
        } else {
            netword.removeClass("network-top-hide");
            navbar.removeClass("navbar-hide");
        }

        lastScrollTop = st;
        tickingScroll = false;
    });
}, { passive: true });

    // Promote above-the-fold lazy images to eager so LCP is not blocked by placeholders.
    (function warmupCriticalImages() {
        var maxWarmup = 18;
        var warmCount = 0;
        var viewportLimit = window.innerHeight * 1.35;
        var candidates = document.querySelectorAll('img.lazy[data-src]');

        for (var i = 0; i < candidates.length && warmCount < maxWarmup; i++) {
            var img = candidates[i];
            var rect = img.getBoundingClientRect();
            var inOrNearViewport = rect.top < viewportLimit && rect.bottom > -80;

            if (!inOrNearViewport) {
                continue;
            }

            var realSrc = img.getAttribute('data-src');
            if (!realSrc) {
                continue;
            }

            // Set loading and fetchpriority first to avoid reflow
            if (warmCount === 0) {
                img.setAttribute('fetchpriority', 'high');
                img.setAttribute('loading', 'eager');
            } else {
                img.setAttribute('loading', 'auto');
                img.setAttribute('fetchpriority', 'auto');
            }
            img.setAttribute('decoding', 'async');
            
            // Swap src after attributes are set to minimize layout shift
            img.src = realSrc;
            img.removeAttribute('data-src');
            img.classList.remove('lazy');
            img.classList.add('loaded');
            warmCount++;
        }
    })();

    $(document).on('click', '.tray-item-trailer', function () {
        var url = $(this).attr('data-url');
        var title = $(this).attr('data-title');
        var link = $(this).attr('data-link');
        var poster = $(this).attr('data-poster');
        var time = Date.now();
        var viewportWidth = Math.max(document.documentElement.clientWidth || 0, window.innerWidth || 0);
        var width = Math.min(Math.max(viewportWidth - 24, 320), 980);
        var height = Math.max(200, Math.round((width * 9) / 16));
        var trailerPlayerId = 'player' + time;
        var trailerInitIntervalId = null;

        function getVideoSourceType(fileUrl) {
            var lowerUrl = (fileUrl || '').toLowerCase();
            return lowerUrl.indexOf('.m3u8') !== -1
                ? 'application/x-mpegURL'
                : 'video/mp4';
        }

        function initTrailerPlayer(playerId, fileUrl, sourceType) {
            if (typeof videojs === 'undefined') {
                return false;
            }
            if (typeof videojs.getPlayer === 'function') {
                var existing = videojs.getPlayer(playerId);
                if (existing) {
                    existing.dispose();
                }
            }

            var player = videojs(playerId, {
                controls: true,
                autoplay: false,
                preload: 'auto',
                responsive: true,
                fluid: true,
                aspectRatio: '16:9',
                playbackRates: [0.75, 1, 1.25, 1.5, 2],
                controlBar: {
                    skipButtons: {
                        backward: 10,
                        forward: 10
                    },
                    volumePanel: {
                        inline: false
                    }
                }
            });

            player.src({
                src: fileUrl,
                type: sourceType
            });

            player.ready(function () {
                player.currentTime(0);
            });

            return true;
        }

        function cleanupTrailerPlayer() {
            if (trailerInitIntervalId) {
                clearInterval(trailerInitIntervalId);
                trailerInitIntervalId = null;
            }

            if (typeof videojs === 'undefined') {
                return;
            }

            $('#pgwModal .video-js').each(function () {
                var playerNodeId = $(this).attr('id');
                if (!playerNodeId) {
                    return;
                }
                try {
                    var player = (typeof videojs.getPlayer === 'function')
                        ? videojs.getPlayer(playerNodeId)
                        : videojs(playerNodeId);
                    if (player && typeof player.dispose === 'function') {
                        player.dispose();
                    }
                } catch (e) {
                    // Ignore dispose errors from stale/removed nodes.
                }
            });
        }

        function mountTrailerPlayer(fileUrl) {
            var sourceType = getVideoSourceType(fileUrl);
            cleanupTrailerPlayer();
            updateModal(loadPlayer(trailerPlayerId, poster, height));

            var attempts = 0;
            var maxAttempts = 30;
            trailerInitIntervalId = setInterval(function () {
                attempts += 1;
                if (initTrailerPlayer(trailerPlayerId, fileUrl, sourceType)) {
                    clearInterval(trailerInitIntervalId);
                    trailerInitIntervalId = null;
                    return;
                }
                if (attempts >= maxAttempts) {
                    clearInterval(trailerInitIntervalId);
                    trailerInitIntervalId = null;
                    $('#modal-content').html('<div style="padding:12px;">Player is loading slowly, please close and open trailer again.</div>');
                }
            }, 120);
        }

        $.pgwModal({
            title: title,
            maxWidth: width,
            height: 'auto',
            content: '<div id="modal-content" class="trailer-modal-content" style="text-align:center;">Loading in progress...</div>',
            closeContent: '[X]'
        });

        if (url.endsWith('.mp4') || url.endsWith('.m3u8')) {
            setTimeout(function () {
                mountTrailerPlayer(url);
            }, 100);
        } else {
            $.ajax({
                url: '/ajax/trailer',
                type: 'POST',
                data: { url: url },
                dataType: 'json',
                success: function (response) {
                    if (response.type === "vip") {
                        if (response.file && (response.file.toLowerCase().indexOf('.mp4') !== -1 || response.file.toLowerCase().indexOf('.m3u8') !== -1)) {
                            mountTrailerPlayer(response.file);
                        } else {
                            updateModal(loadIframeVip(response.file));
                        }
                    } else {
                        updateModal(loadIframeFree(response.file));
                    }
                },
            });
        }

        function loadPlayer(playerId, playerPoster, playerHeight) {
            return '<div class="trailer-player-wrap">' +
                '<video id="' + playerId + '" class="video-js vjs-big-play-centered vjs-sextb-skin" controls preload="auto" playsinline poster="' + playerPoster + '"></video>' +
                '</div>' +
                '<br><center><a href="' + link + '" title="' + title + '">' +
                '<button class="btn-success btn_player"><i class="fa fa-play-circle"></i> Watch Now</button>' +
                '</a></center>';
        }

        function loadIframeFree(url) {
            return '<iframe src="' + url + '?poster=' + poster + '" width="100%" height="' + height + '" frameborder="0" allowfullscreen></iframe>' +
                '<div style="text-align: center;background: #514746; padding: 3px;"><span style="background-image: url(/images/bg_text.gif);">Upgrade VIP member remove Ad-free on site.</span></div>' +
                '<div>' +  
                '<a href="/user/upgrade" title="Upgrade Member"><button class="btn-success btn_player" style="float: left; margin: 10px 0px;><i class="fa fa-play-circle"></i> Upgrade V.I.P</button></a>'+ 
                '<a href="' + link + '" title="' + title + '" style="float: right; margin: 10px 0px;"><button class="btn-success btn_player"><i class="fa fa-play-circle"></i> Watch Free</button></a>'+ 
                '</div>';
        }

        function loadIframeVip(url) {
            return '<iframe src="' + url + '?poster=' + poster + '" width="100%" height="' + height + '" frameborder="0" allowfullscreen></iframe>' +
                '<div style="text-align:center; margin: 10px 0px;">' +  
                '<a href="' + link + '" title="' + title + '"><button class="btn-success btn_player"><i class="fa fa-play-circle"></i> Watch Now</button></a>'+ 
                '</div>';
        }

function xorDecrypt(encoded, key) {
    if (!encoded || !key) return '';
    var decoded = atob(encoded);
    var result = '';
    for (var i = 0; i < decoded.length; i++) {
        result += String.fromCharCode(decoded.charCodeAt(i) ^ key.charCodeAt(i % key.length));
    }
    return result;
}

function validateEmail(email) {
    const re = /^(([^<>()[\]\\.,;:\s@"]+(\.[^<>()[\]\\.,;:\s@"]+)*)|(".+"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;
    return re.test(String(email).toLowerCase());
}

function buildCaptchaUrl(baseUrl) {
    var source = baseUrl || '/captcha';
    return source + '?ver=' + Date.now() + Math.floor(Math.random() * 10000);
}

function refreshCaptchaInScope(scope) {
    var $scope = scope ? $(scope) : $(document);
    $scope.find('img.js-captcha-image').each(function () {
        var $img = $(this);
        var baseUrl = $img.attr('data-captcha-src') || '/captcha';
        $img.attr('src', buildCaptchaUrl(baseUrl));
        $img.attr('data-captcha-loaded', '1');
    });
}

    $scope.find('img.js-captcha-image').each(function () {

        if ($img.attr('data-captcha-loaded') !== '1') {
            var baseUrl = $img.attr('data-captcha-src') || '/captcha';
            $img.attr('src', buildCaptchaUrl(baseUrl));
            $img.attr('data-captcha-loaded', '1');
        }
    });
}

function whenRecaptchaReady(callback) {
    if (typeof callback !== 'function') {
        return;
    }
    if (typeof window.turnstile !== 'undefined' && typeof window.turnstile.render === 'function') {
        callback();
        return;
    }
    setTimeout(function () {
        whenRecaptchaReady(callback);
    }, 120);
}

function normalizeTurnstileSize(size) {
    var value = $.trim(String(size || '')).toLowerCase();
    if (value === 'compact' || value === 'flexible' || value === 'invisible') {
        return value;
    }
    return 'normal';
}

function normalizeTurnstileAppearance(appearance) {
    var value = $.trim(String(appearance || '')).toLowerCase();
    if (value === 'execute' || value === 'interaction-only') {
        return value;
    }
    return 'always';
}

function normalizeTurnstileExecution(execution, size) {
    var value = $.trim(String(execution || '')).toLowerCase();
    if (value === 'execute' || value === 'render') {
        return value;
    }
    return size === 'invisible' ? 'execute' : 'render';
}

function ensureRecaptchaWithRetry(scope) {
    var $scope = scope ? $(scope) : $(document);
    ensureRecaptchaInScope($scope);
    setTimeout(function () {
        ensureRecaptchaInScope($scope);
    }, 180);
    setTimeout(function () {
        ensureRecaptchaInScope($scope);
    }, 650);
}

function waitRecaptchaToken($scope, tokenResolver, onReady, onFail, maxAttempts) {
    var attempt = 0;
    var limit = (typeof maxAttempts === 'number' && maxAttempts > 0) ? maxAttempts : 35;
    var intervalId = setInterval(function () {
        var nextToken = $.trim(tokenResolver($scope) || '');
        if (nextToken !== '') {
            clearInterval(intervalId);
            onReady(nextToken);
            return;
        }

        attempt += 1;
        if (attempt >= limit) {
            clearInterval(intervalId);
            if (typeof onFail === 'function') {
                onFail();
            }
        }
    }, 120);
}

function hasTurnstileFrame($el) {
    if (!$el || !$el.length) {
        return false;
    }
    return $el.find('iframe').length > 0;
}

function ensureRecaptchaInScope(scope) {
    var $scope = scope ? $(scope) : $(document);
    whenRecaptchaReady(function () {
        $scope.find('.js-recaptcha-box').each(function () {
            var $el = $(this);
            var widgetId = $.trim(String($el.attr('data-widget-id') || ''));
            var needRender = (widgetId === '' || !hasTurnstileFrame($el) || $el.attr('data-render-error') === '1');
            if (needRender) {
                var siteKey = $el.attr('data-sitekey');
                if (!siteKey) {
                    return;
                }
                try {
                    if (widgetId !== '' && typeof window.turnstile !== 'undefined' && typeof window.turnstile.remove === 'function') {
                        try {
                            window.turnstile.remove(widgetId);
                        } catch (removeErr) {}
                    }
                    var theme = $el.attr('data-theme') || 'dark';
                    var size = normalizeTurnstileSize($el.attr('data-size'));
                    var appearance = normalizeTurnstileAppearance($el.attr('data-appearance'));
                    var execution = normalizeTurnstileExecution($el.attr('data-execution'), size);
                    var options = {
                        sitekey: siteKey,
                        theme: theme,
                        size: size,
                        appearance: appearance,
                        'expired-callback': function () {
                            try {
                                window.turnstile.reset($el.attr('data-widget-id'));
                            } catch (err) {}
                        },
                        'error-callback': function () {
                            try {
                                window.turnstile.reset($el.attr('data-widget-id'));
                            } catch (err) {}
                        }
                    };

                    if (execution === 'execute') {
                        options.execution = 'execute';
                    }

                    widgetId = window.turnstile.render(this, options);
                    $el.attr('data-widget-id', String(widgetId));
                    $el.removeAttr('data-render-error');
                } catch (e) {
                    $el.attr('data-render-error', '1');
                }
            }
        });
    });
}

function requestRecaptchaToken($scope, tokenResolver, onReady, onFail) {
    var token = $.trim(tokenResolver($scope) || '');
    if (token !== '') {
        onReady(token);
        return;
    }

    var $widget = $scope.find('.js-recaptcha-box').first();
    var executionMode = normalizeTurnstileExecution($widget.attr('data-execution'), normalizeTurnstileSize($widget.attr('data-size')));

    if (executionMode !== 'execute') {
        ensureRecaptchaWithRetry($scope);
        waitRecaptchaToken($scope, tokenResolver, onReady, onFail, 80);
        return;
    }

    var widgetId = $.trim(String($widget.attr('data-widget-id') || ''));
    if (!widgetId || typeof window.turnstile === 'undefined' || typeof window.turnstile.execute !== 'function') {
        ensureRecaptchaWithRetry($scope);
        if (typeof onFail === 'function') {
            onFail();
        }
        return;
    }

    try {
        window.turnstile.execute(widgetId);
    } catch (e) {
        if (typeof onFail === 'function') {
            onFail();
        }
        return;
    }

    waitRecaptchaToken($scope, tokenResolver, onReady, onFail, 50);
}

        return getRecaptchaTokenInForm(ctx);
    }, onReady, onFail);
}

        return getRecaptchaTokenInScope(ctx);
    }, onReady, onFail);
}

function getRecaptchaTokenInForm($form) {
    var token = '';
    if (typeof window.turnstile !== 'undefined' && typeof window.turnstile.getResponse === 'function') {
        var widgetId = $form.find('.js-recaptcha-box').attr('data-widget-id');
        if (widgetId !== undefined && widgetId !== null && widgetId !== '') {
            token = $.trim(window.turnstile.getResponse(widgetId) || '');
        }
    }
    if (!token) {
        token = $.trim($form.find('input[name="cf-turnstile-response"]').val() || '');
    }
    if (!token) {
        token = $.trim($form.find('textarea[name="g-recaptcha-response"]').val() || '');
    }
    return token;
}

$(document).on('click', '.favorite-movie, .favorite-actress, .favorite-studio, .favorite-director, .favorite-label', function(e){
    e.preventDefault();

    var trigger = $(this);
    var id = parseInt(trigger.attr('data-id'), 10) || 0;
    var name = trigger.attr('data-name') || '';
    var type = (trigger.attr('data-type') || '').trim();
    if (!type || id <= 0) {
        return;
    }

    var isActiveNow;
    if (type === 'movie') {
        isActiveNow = trigger.hasClass('btn-player-active');
    } else if (type === 'actress') {
        isActiveNow = trigger.hasClass('heart-active');
    } else {
        isActiveNow = trigger.hasClass('heart-active');
    }
    var desiredAction = isActiveNow ? 'remove' : 'add';

    var lockKey = type + ':' + id;
    if (window.__favClickLocks[lockKey]) {
        return;
    }

    var targetSelector = (type === 'movie')
        ? '.favorite-movie'
        : '.favorite-' + type + '[data-id="' + id + '"]';
    var targets = $(targetSelector);
    if (!targets.length) {
        targets = trigger;
    }

    if (targets.filter(function(){ return $(this).data('pending') === true; }).length > 0) {
        return;
    }

    var now = Date.now();
    var cooldownUntil = parseInt(targets.first().data('cooldownUntil') || 0, 10);
    if (!isNaN(cooldownUntil) && cooldownUntil > now) {
        return;
    }

    let isAdBlocked = typeof window.pJGQYAfPSq === 'function' ? window.pJGQYAfPSq() : false;
    if (isAdBlocked) {
        $.pgwModal({
            target: '#noteads',
            title: 'AdBlock Detected!',
            maxWidth: 400,
            closeOnBackgroundClick : true,
            closeContent : '[X]'
        });
        return;
    }

    window.__favClickLocks[lockKey] = true;
    targets.data('pending', true).addClass('is-pending');

    // Optimistic UI: update class immediately (no toast yet to avoid clutter)
    var willBeActive = desiredAction === 'add';
    if (type === 'movie') {
        targets.toggleClass('btn-player-active', willBeActive);
    } else if (type === 'actress') {
        targets.toggleClass('heart-active', willBeActive);
    } else {
        targets.toggleClass('heart-active', willBeActive);
    }

    $.post('/user/addfav', {id: id, name: name, type: type, page: 1, action: desiredAction}, function(res) {
        if (res && res.status === true) {
            if (typeof res.count !== 'undefined') {
                var favCount = parseInt(res.count, 10);
                if (isNaN(favCount) || favCount < 0) {
                    favCount = 0;
                }
                $('#favoriteCount').html(favCount > 0 ? kFormatter(favCount) : 0);
            }

            var isActive = (typeof res.active !== 'undefined')
                ? !!res.active
                : (res.type === 'add');

            if (type === 'movie') {
                targets.toggleClass('btn-player-active', isActive);
            } else if (type === 'actress') {
                targets.toggleClass('heart-active', isActive);
            } else {
                targets.toggleClass('heart-active', isActive);
            }

            if (isActive) {
                $.toast({
                    text: res.msg,
                    position: 'mid-center',
                    icon: 'success',
                    stack: false
                });
            } else {
                $.toast({
                    text: res.msg,
                    position: 'mid-center',
                    icon: 'warning',
                    stack: false
                });
            }
            return;
        }

        // Revert optimistic UI on error
        if (type === 'movie') {
            targets.toggleClass('btn-player-active', !willBeActive);
        } else if (type === 'actress') {
            targets.toggleClass('heart-active', !willBeActive);
        } else {
            targets.toggleClass('heart-active', !willBeActive);
        }

        if (res && res.rate_limited) {
            $.toast({
                text: res.msg || 'Too many requests. Please wait and try again.',
                position: 'mid-center',
                icon: 'warning',
                stack: false
            });
        } else if (res && res.msg && String(res.msg).toLowerCase().indexOf('login') === -1) {
            $.toast({
                heading: 'Error!!!',
                text: res.msg,
                position: 'mid-center',
                icon: 'error',
                stack: false
            });
        } else {
            $.pgwModal({
                target: '#modalMember',
                title: 'Login To Favorite',
                maxWidth: 400,
                closeOnBackgroundClick : false,
                closeContent : '[X]'
            });
        }
    }).always(function() {
        targets.data('pending', false).removeClass('is-pending');
        var unlockAt = Date.now() + 220;
        targets.data('cooldownUntil', unlockAt);
        setTimeout(function() {
            targets.data('cooldownUntil', 0);
            window.__favClickLocks[lockKey] = false;
        }, 260);
    });
});

function viewed_format(num) {
     if (num >= 1000000000) {
        return (num / 1000000000).toFixed(1).replace(/\.0$/, '') + 'G';
     }
     if (num >= 1000000) {
        return (num / 1000000).toFixed(1).replace(/\.0$/, '') + 'M';
     }
     if (num >= 1000) {
        return (num / 1000).toFixed(1).replace(/\.0$/, '') + 'K';
     }
     return num;
}

// SetCookie
function setCookie(cname, cvalue, exdays) {
    const d = new Date();
    d.setTime(d.getTime() + (exdays*24*60*60*1000));
    let expires = "expires="+ d.toUTCString();
    document.cookie = cname + "=" + cvalue + ";" + expires + ";path=/";
}

// GetCookie
function getCookie(cname) {
    let name = cname + "=";
    let decodedCookie = decodeURIComponent(document.cookie);
    let ca = decodedCookie.split(';');
    for(let i = 0; i <ca.length; i++) {
        let c = ca[i];
        while (c.charAt(0) == ' ') {
            c = c.substring(1);
        }
        if (c.indexOf(name) == 0) {
            return c.substring(name.length, c.length);
        }
    }
    return "";
}

function relatedMV(item) {
    let html = '';
    $.each(item, function(index, element) {
        let ribbon = '', border = '', quality = '<span class="sd">SD</span>', offensive = '', figcaption = '', code = element['guid'] || element['code'];
        
        if (element['genre'] === '2' && element['subtitle'] === '0') {

            border = ' tray-item-uncensored';

            ribbon = '<span class="tray-item-sub">English Subtitle</span>';
            border = ' tray-item-subtitle';

            border = ' tray-item-sub-uncen';
        }

            figcaption = '<figcaption>Offensive</figcaption>';

                : '';

                            <div class="tray-item-title">${element['name']}</div>

                </div>`;
    });

}

function checkAndToggleText(selector, maxLength = 550) {
    document.querySelectorAll(selector).forEach(container => {
        let fullText = container.querySelector(".full-text-desc"),
            shortText = container.querySelector(".short-text"),
            toggleBtn = container.querySelector(".toggle-btn");

        if (fullText.textContent.length > maxLength) {
            shortText.textContent = fullText.textContent.slice(0, maxLength) + "...";
            fullText.style.display = "none";
            toggleBtn.style.display = "inline";
        } else {
            shortText.textContent = fullText.textContent;
            toggleBtn.style.display = "none";
        }

        toggleBtn.onclick = () => {
            let isHidden = fullText.style.display === "none";
            fullText.style.display = isHidden ? "inline" : "none";
            shortText.style.display = isHidden ? "none" : "inline";
            toggleBtn.textContent = isHidden ? "â† Hidden" : "â†’ Show more";
        };
    });
}
